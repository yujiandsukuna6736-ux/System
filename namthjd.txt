local function getPlayerExp(target)
    local stats = target:FindFirstChild("leaderstats")
    if stats then
        local expNode = stats:FindFirstChild("EXP") 
                     or stats:FindFirstChild("Exp") 
                     or stats:FindFirstChild("Experience")
                     or stats:FindFirstChild("Xp")
                     or stats:FindFirstChild("Cấp")
        if expNode then
            return tostring(expNode.Value)
        end
    end
    return "0"
end

local function updatePlayerList()
    for _, item in ipairs(playerScroll:GetChildren()) do
        if item:IsA("Frame") then
            item:Destroy()
        end
    end
    
    for _, targetPlayer in ipairs(Players:GetPlayers()) do
        if targetPlayer ~= player then
            local pRow = Instance.new("Frame")
            pRow.Size = UDim2.new(1, -5, 0, 28)
            pRow.BackgroundTransparency = 1
            pRow.Parent = playerScroll

            local pExpVal = getPlayerExp(targetPlayer)
            local pBtn = Instance.new("TextButton")
            pBtn.Size = UDim2.new(0, 125, 1, 0)
            pBtn.BackgroundColor3 = Color3.fromRGB(240, 120, 0)
            pBtn.Text = "👤 " .. targetPlayer.DisplayName .. " [" .. pExpVal .. "]"
            pBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
            pBtn.TextSize = 10
            pBtn.Font = Enum.Font.SourceSansBold
            pBtn.Parent = pRow

            local pCorner = Instance.new("UICorner")
            pCorner.CornerRadius = UDim.new(0, 4)
            pCorner.Parent = pBtn

            local lockBtn = Instance.new("TextButton")
            lockBtn.Size = UDim2.new(0, 70, 1, 0)
            lockBtn.Position = UDim2.new(0, 130, 0, 0)
            lockBtn.BackgroundColor3 = Color3.fromRGB(200, 100, 0)
            lockBtn.Text = "🎯 Định vị"
            lockBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
            lockBtn.TextSize = 11
            lockBtn.Font = Enum.Font.SourceSansBold
            lockBtn.Parent = pRow

            local lCorner = Instance.new("UICorner")
            lCorner.CornerRadius = UDim.new(0, 4)
            lCorner.Parent = lockBtn

            pBtn.MouseButton1Click:Connect(function()
                if targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
                    local myChar = player.Character
                    if myChar and myChar:FindFirstChild("HumanoidRootPart") then
                        myChar.HumanoidRootPart.CFrame = targetPlayer.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, 2)
                    end
                end
            end)

            lockBtn.MouseButton1Click:Connect(function()
                if trackedTarget == targetPlayer then
                    trackedTarget = nil
                    lockBtn.BackgroundColor3 = Color3.fromRGB(200, 100, 0)
                    lockBtn.Text = "🎯 Định vị"
                    trackerStatus.Text = "🎯 Định vị: Đã tắt"
                else
                    trackedTarget = targetPlayer
                    lockBtn.BackgroundColor3 = Color3.fromRGB(40, 180, 40)
                    lockBtn.Text = "✅ Đang dò"
                    trackerStatus.Text = "🎯 Đang theo dõi: " .. targetPlayer.Name
                end
            end)
        end
    end
end

flingButton.MouseButton1Click:Connect(function()

    if flingEnabled then
        flingButton.Text = "🌪️ Chạm văng: ON"
        flingButton.BackgroundColor3 = Color3.fromRGB(40, 180, 40)
        
        flingConnection = RunService.Stepped:Connect(function()
            local character = player.Character
            if character and character:FindFirstChild("HumanoidRootPart") then
                local rootPart = character.HumanoidRootPart
                local angularVelocity = rootPart:FindFirstChild("FlingAngularVelocity")
                
                if not angularVelocity then
                    angularVelocity = Instance.new("BodyAngularVelocity")
                    angularVelocity.Name = "FlingAngularVelocity"
                    angularVelocity.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
                    angularVelocity.AngularVelocity = Vector3.new(0, 99999, 0)
                    angularVelocity.Parent = rootPart
                end
            end
        end)
    else
        flingButton.Text = "🌪️ Chạm văng: OFF"
        flingButton.BackgroundColor3 = Color3.fromRGB(220, 80, 0)
        if flingConnection then
            flingConnection:Disconnect()
            flingConnection = nil
        end
        
        local character = player.Character
        if character and character:FindFirstChild("HumanoidRootPart") then
            local rootPart = character.HumanoidRootPart
            if rootPart:FindFirstChild("FlingAngularVelocity") then
                rootPart.FlingAngularVelocity:Destroy()
            end
        end
    end
end)

antiLagButton.MouseButton1Click:Connect(function()
    if antiLagEnabled then return end
    antiLagEnabled = true
    
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 9e9
    
    for _, v in ipairs(Lighting:GetChildren()) do
        if v:IsA("PostEffect") then v.Enabled = false end
    end

    local function optimizeObject(obj)
        if obj:IsA("BasePart") then
            obj.Material = Enum.Material.SmoothPlastic
            obj.CastShadow = false
        elseif obj:IsA("Decal") or obj:IsA("Texture") then
            obj.Transparency = 1
        elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Smoke") or obj:IsA("Fire") or obj:IsA("Sparkles") then
            obj.Enabled = false
        end
    end

    for _, v in ipairs(game.Workspace:GetDescendants()) do optimizeObject(v) end
    game.Workspace.DescendantAdded:Connect(function(v)
        if antiLagEnabled then optimizeObject(v) end
    end)

    antiLagButton.Text = "🚀 Đã Giảm Lag! (ON)"
    antiLagButton.BackgroundColor3 = Color3.fromRGB(40, 180, 40)
end)

noclipButton.MouseButton1Click:Connect(function()

        noclipButton.Text = "Noclip: ON"
        noclipButton.BackgroundColor3 = Color3.fromRGB(40, 180, 40)
        noclipConnection = RunService.Stepped:Connect(function()
            local character = player.Character

                    if part:IsA("BasePart") then part.CanCollide = false end
                end

        end)
    else

        noclipButton.BackgroundColor3 = Color3.fromRGB(220, 80, 0)
        if noclipConnection then
            noclipConnection:Disconnect()

        end

        if character then
            for _, part in ipairs(character:GetDescendants()) do
                if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
                    part.CanCollide = true
                end
            end
        end
    end
end)
